
#include <stdio.h>

void rightanglewithnumbers(int n);
void pyramidwithstars(int n);
void pyramidwithAlphabets(int n);
void diamondwithnumbers(int n);

int main() {
    int n;

    printf("Enter the number of rows: ");
    scanf("%d", &n);

    rightanglewithnumbers(n);
    pyramidwithstars(n);
    pyramidwithAlphabets(n);
    diamondwithnumbers(n);

    return 0;
}

void rightanglewithnumbers(int n) {
    int i, j;
    printf("Right angle triangle (1 22 333...)\n");
    for (i = 1; i <= n; i++) {
        for (j = 1; j <= i; j++) {
            printf("%d", i);
        }
        printf("\n");
    }
}

void pyramidwithstars(int n) {
    int i, j, space;
    printf("Pyramid with *\n");
    for (i = 1; i <= n; i++) {
        for (space = 1; space <= n - i; space++) {
            printf(" ");
        }
        for (j = 1; j <= 2 * i - 1; j++) {
            printf("*");
        }
        printf("\n");
    }
}

void pyramidwithAlphabets(int n) {
    int i, j, space;
    char ch;
    printf("Pyramid with alphabets\n");
    for (i = 1; i <= n; i++) {
        ch = 'A';
        for (space = 1; space <= n - i; space++) {
            printf(" ");
        }
        for (j = 1; j <= i; j++) {
            printf("%c", ch++);
        }
        printf("\n");
    }
}

void diamondwithnumbers(int n) {
    int i, j, space;
    printf("Diamond with numbers\n");
    // Upper half
    for (i = 1; i <= n; i++) {
        for (space = 1; space <= n - i; space++) {
            printf(" ");
        }
        for (j = 1; j <= 2 * i - 1; j++) {
            printf("%d", i);
        }
        printf("\n");
    }

    // Lower half
    for (i = n - 1; i >= 1; i--) {
        for (space = 1; space <= n - i; space++) {
            printf(" ");
        }
        for (j = 1; j <= 2 * i - 1; j++) {
            printf("%d", i);
        }
        printf("\n");
    }
}
